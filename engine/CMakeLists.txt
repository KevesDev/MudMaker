cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" MUDMAKER_VERSION LIMIT_COUNT 1)
project(mudmaker_engine VERSION ${MUDMAKER_VERSION} LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The native core currently supports Windows/MSVC only.")
endif()
option(MUDMAKER_ENABLE_ASAN "Instrument the focused native identity library and tests" OFF)

add_library(mudmaker_engine_core STATIC src/ecs/identity.cpp src/ecs/component_schema.cpp src/ecs/world.cpp)
target_include_directories(mudmaker_engine_core PUBLIC include PRIVATE src)
target_link_libraries(mudmaker_engine_core PRIVATE bcrypt)
add_executable(mudmaker_identity_tests tests/identity_allocator_tests.cpp)
target_link_libraries(mudmaker_identity_tests PRIVATE mudmaker_engine_core)
target_include_directories(mudmaker_identity_tests PRIVATE src)
add_executable(mudmaker_component_schema_tests tests/component_schema_tests.cpp)
target_link_libraries(mudmaker_component_schema_tests PRIVATE mudmaker_engine_core)
add_executable(mudmaker_world_tests tests/world_tests.cpp)
target_link_libraries(mudmaker_world_tests PRIVATE mudmaker_engine_core)
foreach(target mudmaker_engine_core mudmaker_identity_tests mudmaker_component_schema_tests mudmaker_world_tests)
  target_compile_features(${target} PUBLIC cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MUDMAKER_ENABLE_ASAN)
    target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
  endif()
endforeach()
if(MUDMAKER_ENABLE_ASAN)
  target_link_options(mudmaker_identity_tests PRIVATE /INCREMENTAL:NO /DEBUG)
  target_link_options(mudmaker_component_schema_tests PRIVATE /INCREMENTAL:NO /DEBUG)
  target_link_options(mudmaker_world_tests PRIVATE /INCREMENTAL:NO /DEBUG)
endif()
enable_testing()
add_test(NAME ecs_identity_allocator COMMAND mudmaker_identity_tests)
add_test(NAME ecs_component_schema COMMAND mudmaker_component_schema_tests)
add_test(NAME ecs_world_mutation COMMAND mudmaker_world_tests)
get_filename_component(MUDMAKER_COMPILER_RUNTIME_DIR "${CMAKE_CXX_COMPILER}" DIRECTORY)
set_tests_properties(ecs_identity_allocator ecs_component_schema ecs_world_mutation PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${MUDMAKER_COMPILER_RUNTIME_DIR}")

option(MUDMAKER_ENABLE_ENGINE_WIRE "Build the pinned engine-control codec foundation" OFF)
if(MUDMAKER_ENABLE_ENGINE_WIRE)
  set(WIRE_EXPECTED_TARGET x64-windows-cxx20)
  if(MUDMAKER_ENABLE_ASAN)
    set(WIRE_EXPECTED_TARGET x64-windows-cxx20-asan)
  endif()
  if(NOT VCPKG_TARGET_TRIPLET STREQUAL WIRE_EXPECTED_TARGET OR NOT VCPKG_HOST_TRIPLET STREQUAL "x64-windows-cxx20")
    message(FATAL_ERROR "Engine wire target/host triplets conflict with the approved sanitizer ABI")
  endif()
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  find_package(protobuf CONFIG REQUIRED)
  set(WIRE_RUNTIME_CACHE "${CMAKE_CURRENT_SOURCE_DIR}/.deps/vcpkg/buildtrees/protobuf/${VCPKG_TARGET_TRIPLET}-rel/CMakeCache.txt")
  if(VCPKG_INSTALL_OPTIONS MATCHES "(^|;)--x-buildtrees-root=([^;]+)")
    set(WIRE_RUNTIME_CACHE "${CMAKE_MATCH_2}/protobuf/${VCPKG_TARGET_TRIPLET}-rel/CMakeCache.txt")
  endif()
  if(EXISTS "${WIRE_RUNTIME_CACHE}")
    file(STRINGS "${WIRE_RUNTIME_CACHE}" WIRE_RUNTIME_FLAGS REGEX "^CMAKE_CXX_FLAGS:STRING=")
    file(STRINGS "${WIRE_RUNTIME_CACHE}" WIRE_RUNTIME_STANDARD REGEX "^CMAKE_CXX_STANDARD(:[^=]*)?=20$")
    if(NOT WIRE_RUNTIME_STANDARD OR WIRE_RUNTIME_FLAGS MATCHES "PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT")
      message(FATAL_ERROR "Protobuf dependency cache conflicts with the selected C++20 ABI")
    endif()
    if(MUDMAKER_ENABLE_ASAN AND NOT WIRE_RUNTIME_FLAGS MATCHES "fsanitize=address")
      message(FATAL_ERROR "Protobuf dependency cache lacks the required ASan instrumentation")
    endif()
  endif()
  if(NOT protobuf_VERSION VERSION_EQUAL "33.6.0")
    message(FATAL_ERROR "Engine wire requires pinned protobuf CMake package33.6.0, found ${protobuf_VERSION}")
  endif()
  file(STRINGS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/include/google/protobuf/runtime_version.h" WIRE_RUNTIME_PIN REGEX "^#define PROTOBUF_OSS_VERSION 6033006$")
  if(NOT WIRE_RUNTIME_PIN)
    message(FATAL_ERROR "Engine wire requires PROTOBUF_OSS_VERSION6033006 from the pinned port")
  endif()
  include(CheckCXXSourceCompiles)
  set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
  set(CMAKE_REQUIRED_LIBRARIES protobuf::libprotobuf)
  set(CMAKE_REQUIRED_FLAGS "/W4 /WX /permissive-")
  if(MUDMAKER_ENABLE_ASAN)
    string(APPEND CMAKE_REQUIRED_FLAGS " /fsanitize=address /Zi")
    set(CMAKE_REQUIRED_LINK_OPTIONS /INCREMENTAL:NO /DEBUG)
  endif()
  check_cxx_source_compiles("#include <google/protobuf/port.h>\nint main() { const auto* value = &google::protobuf::internal::fixed_address_empty_string; return static_cast<int>(value->get().size()); }" MUDMAKER_PROTOBUF_CXX20_ABI)
  unset(CMAKE_REQUIRED_LIBRARIES)
  unset(CMAKE_REQUIRED_FLAGS)
  unset(CMAKE_REQUIRED_LINK_OPTIONS)
  if(NOT MUDMAKER_PROTOBUF_CXX20_ABI)
    message(FATAL_ERROR "Pinned protobuf runtime does not provide its required C++20 ABI")
  endif()
  set(MUDMAKER_PINNED_PROTOC "${VCPKG_INSTALLED_DIR}/${VCPKG_HOST_TRIPLET}/tools/protobuf/protoc.exe")
  execute_process(COMMAND "${MUDMAKER_PINNED_PROTOC}" --version OUTPUT_VARIABLE MUDMAKER_PROTOC_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE MUDMAKER_PROTOC_RESULT)
  if(NOT MUDMAKER_PROTOC_RESULT EQUAL 0 OR NOT MUDMAKER_PROTOC_VERSION STREQUAL "libprotoc 33.6")
    message(FATAL_ERROR "Engine wire requires the manifest's protoc33.6, found ${MUDMAKER_PROTOC_VERSION}")
  endif()
  set(WIRE_CONTRACTS "${CMAKE_CURRENT_SOURCE_DIR}/../contracts")
  set(WIRE_GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
  file(MAKE_DIRECTORY "${WIRE_GENERATED}")
  set(WIRE_FILES engine/v1/mudmaker_engine_v1 scripting/v1/mudmaker_scripting_v1)
  set(WIRE_SOURCES)
  foreach(file IN LISTS WIRE_FILES)
    list(APPEND WIRE_SOURCES "${WIRE_GENERATED}/${file}.pb.cc" "${WIRE_GENERATED}/${file}.pb.h")
  endforeach()
  add_custom_command(OUTPUT ${WIRE_SOURCES} "${WIRE_GENERATED}/mudmaker_engine_v1.descriptor.pb"
    COMMAND "${MUDMAKER_PINNED_PROTOC}" "--proto_path=${WIRE_CONTRACTS}" "--cpp_out=${WIRE_GENERATED}" "--descriptor_set_out=${WIRE_GENERATED}/mudmaker_engine_v1.descriptor.pb" --include_imports "${WIRE_CONTRACTS}/engine/v1/mudmaker_engine_v1.proto" "${WIRE_CONTRACTS}/scripting/v1/mudmaker_scripting_v1.proto"
    DEPENDS "${WIRE_CONTRACTS}/engine/v1/mudmaker_engine_v1.proto" "${WIRE_CONTRACTS}/scripting/v1/mudmaker_scripting_v1.proto" VERBATIM)
  add_library(mudmaker_engine_proto STATIC ${WIRE_SOURCES})
  target_include_directories(mudmaker_engine_proto SYSTEM PUBLIC "${WIRE_GENERATED}")
  target_link_libraries(mudmaker_engine_proto PUBLIC protobuf::libprotobuf)
  target_compile_features(mudmaker_engine_proto PUBLIC cxx_std_20)
  if(MUDMAKER_ENABLE_ASAN)
    # Generated accessors share message/STL ABI with instrumented owned consumers.
    target_compile_options(mudmaker_engine_proto PRIVATE /fsanitize=address /Zi)
  endif()
  add_library(mudmaker_engine_wire STATIC src/wire/codec.cpp src/wire/descriptor.cpp)
  target_include_directories(mudmaker_engine_wire PUBLIC include)
  target_link_libraries(mudmaker_engine_wire PUBLIC mudmaker_engine_proto mudmaker_engine_core)
  add_executable(mudmaker_engine_wire_tests tests/engine_wire_tests.cpp)
  target_link_libraries(mudmaker_engine_wire_tests PRIVATE mudmaker_engine_wire)
  target_compile_definitions(mudmaker_engine_wire_tests PRIVATE "WIRE_POLICY_FILE=\"${WIRE_CONTRACTS}/engine/v1/protocol-policy.json\"" "WIRE_FROZEN_FILE=\"${WIRE_CONTRACTS}/engine/v1/mudmaker_engine_v1.descriptor.pb\"" "WIRE_CURRENT_FILE=\"${WIRE_GENERATED}/mudmaker_engine_v1.descriptor.pb\"")
  foreach(target mudmaker_engine_wire mudmaker_engine_wire_tests)
    target_compile_features(${target} PUBLIC cxx_std_20)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
    if(MUDMAKER_ENABLE_ASAN)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
    endif()
  endforeach()
  if(MUDMAKER_ENABLE_ASAN)
    target_link_options(mudmaker_engine_wire_tests PRIVATE /INCREMENTAL:NO /DEBUG)
  endif()
  add_test(NAME engine_wire_codec COMMAND mudmaker_engine_wire_tests)
  set_tests_properties(engine_wire_codec PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${MUDMAKER_COMPILER_RUNTIME_DIR};PATH=path_list_prepend:${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin")
endif()
